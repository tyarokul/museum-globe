<script src="https://cdn.jsdelivr.net/npm/three@0.160.0/build/three.min.js"></script>
<script src="https://cdn.jsdelivr.net/npm/three@0.160.0/examples/js/controls/OrbitControls.js"></script>

<script>
let scene = new THREE.Scene();

let camera = new THREE.PerspectiveCamera(
  45,
  window.innerWidth / window.innerHeight,
  0.1,
  1000
);

camera.position.z = 3;

let renderer = new THREE.WebGLRenderer({
  antialias: true,
  alpha: true
});

renderer.setSize(window.innerWidth, window.innerHeight);
renderer.setClearColor(0x000000, 0);
document.getElementById("container").appendChild(renderer.domElement);

// Controls
let controls = new THREE.OrbitControls(camera, renderer.domElement);
controls.enableZoom = true;
controls.enablePan = false;
controls.rotateSpeed = 0.4;

// LIGHT
let light = new THREE.AmbientLight(0xffffff, 1);
scene.add(light);

// EARTH TEXTURE (REALISTIC)
let textureLoader = new THREE.TextureLoader();

let earthGeometry = new THREE.SphereGeometry(1, 64, 64);

let earthMaterial = new THREE.MeshStandardMaterial({
  map: textureLoader.load("https://threejs.org/examples/textures/planets/earth_atmos_2048.jpg"),
  roughness: 1,
  metalness: 0
});

let earth = new THREE.Mesh(earthGeometry, earthMaterial);
scene.add(earth);

// HOTSPOTS (MUSEUMS)
const museums = [
  {
    name: "Louvre",
    lat: 48.8606,
    lng: 2.3376,
    url: "https://www.louvre.fr"
  },
  {
    name: "British Museum",
    lat: 51.5194,
    lng: -0.1270,
    url: "https://www.britishmuseum.org"
  },
  {
    name: "Topkapi Palace",
    lat: 41.0115,
    lng: 28.9833,
    url: "https://muze.gen.tr/muze-detay/topkapi"
  }
];

// convert lat/lng to 3D
function latLngToVector3(lat, lng, radius) {
  let phi = (90 - lat) * (Math.PI / 180);
  let theta = (lng + 180) * (Math.PI / 180);

  let x = -radius * Math.sin(phi) * Math.cos(theta);
  let z = radius * Math.sin(phi) * Math.sin(theta);
  let y = radius * Math.cos(phi);

  return new THREE.Vector3(x, y, z);
}

// create markers
let markers = [];

museums.forEach(m => {
  let pos = latLngToVector3(m.lat, m.lng, 1.01);

  let markerGeo = new THREE.SphereGeometry(0.015, 12, 12);
  let markerMat = new THREE.MeshBasicMaterial({ color: 0xffcc00 });

  let marker = new THREE.Mesh(markerGeo, markerMat);
  marker.position.copy(pos);

  marker.userData = m;

  scene.add(marker);
  markers.push(marker);
});

// raycaster
let raycaster = new THREE.Raycaster();
let mouse = new THREE.Vector2();

window.addEventListener("click", (event) => {
  mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
  mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;

  raycaster.setFromCamera(mouse, camera);

  let intersects = raycaster.intersectObjects(markers);

  if (intersects.length > 0) {
    let url = intersects[0].object.userData.url;
    window.open(url, "_blank");
  }
});

// hover label
let label = document.getElementById("label");

window.addEventListener("mousemove", (event) => {
  mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
  mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;

  raycaster.setFromCamera(mouse, camera);

  let intersects = raycaster.intersectObjects(markers);

  if (intersects.length > 0) {
    let obj = intersects[0].object;

    label.style.display = "block";
    label.style.left = event.clientX + "px";
    label.style.top = event.clientY + "px";
    label.innerHTML = obj.userData.name;
  } else {
    label.style.display = "none";
  }
});

// animation
function animate() {
  requestAnimationFrame(animate);

  earth.rotation.y += 0.0008;

  controls.update();
  renderer.render(scene, camera);
}

animate();

window.addEventListener("resize", () => {
  camera.aspect = window.innerWidth / window.innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(window.innerWidth, window.innerHeight);
});
